### Phase 2: Core External Consumer Smoke Test
Execute in an isolated temporary directory outside the monorepo:
```bash
mkdir -p /tmp/smoke_portakal_core && cd /tmp/smoke_portakal_core
dart create -t console-simple .
dart pub add portakal_core:1.0.0-rc.1
```
*Code validation*:
```dart
import 'dart:typed_data';
import 'package:portakal_core/portakal_core.dart';

void main() {
  final escposBytes = (EscPosPrinter()..initialize()..textLine('Smoke Test')..cut()).toBytes();
  final tscBytes = (TscPrinter()..sizeDots(800, 1200)..cls()..text(x: 10, y: 10, text: 'Smoke')..print()).toBytes();
  final labelAST = label(const LabelConfig(width: 50, height: 30))..text('AST Smoke', const TextOptions(x: 5, y: 5));
  final zplBytes = zpl.compile(labelAST);

  assert(escposBytes is Uint8List && escposBytes.isNotEmpty);
  assert(tscBytes is Uint8List && tscBytes.isNotEmpty);
  assert(zplBytes is Uint8List && zplBytes.isNotEmpty);
  print('portakal_core 1.0.0-rc.1 external smoke PASS');
}
```
